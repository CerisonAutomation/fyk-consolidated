export function postAge(minutes: number): string {
	if (minutes < 1) return "now";
	if (minutes < 60) return `${Math.round(minutes)}m`;
	if (minutes < 1440) return `${Math.floor(minutes / 60)}h`;
	return `${Math.floor(minutes / 1440)}d`;
}
